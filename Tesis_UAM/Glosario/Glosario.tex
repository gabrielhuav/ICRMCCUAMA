\newglossaryentry{noticiafalsa}{
	name={Noticias Falsas},
	description={Información deliberadamente fabricada que se presenta como contenido periodístico legítimo, pero que contiene datos falsos, inexactos o engañosos. Su objetivo principal es manipular la opinión pública, influir en decisiones políticas o sociales, o distorsionar la percepción de eventos actuales. Se caracteriza por imitar el formato y estilo de medios de comunicación establecidos, utilizando técnicas periodísticas aparentemente profesionales para ganar credibilidad. A diferencia del fraude digital, su ``éxito'' se mide por el alcance social y la influencia en la opinión pública, no por beneficios económicos directos.}
}

\newglossaryentry{bulo}{
	name={Bulo},
	description={Información falsa que circula ampliamente, especialmente en redes sociales y medios digitales, con el propósito de engañar a la audiencia. A diferencia de las noticias falsas, los bulos pueden no tener formato periodístico y suelen propagarse de manera viral. Incluye rumores, teorías conspirativas, información médica falsa, y contenido que apela más a las emociones que a los hechos verificables.}
}

\newglossaryentry{desinformacion}{
	name={Desinformación},
	description={Difusión intencional de información falsa o engañosa con el propósito específico de manipular la opinión pública, influir en decisiones políticas o sociales, o causar daño. Se caracteriza por ser un proceso activo y deliberado de creación y distribución de contenido falso.}
}

\newglossaryentry{misinformacion}{
	name={Desinformación (Misinformation)},
	description={Información incorrecta que se comparte sin intención maliciosa. A diferencia de la desinformación, quienes comparten misinformación no tienen conocimiento de que la información es falsa y actúan de buena fe.}
}

\newglossaryentry{malinformacion}{
	name={Información Maliciosa (Malinformation)},
	description={Información genuina que se comparte con intención de causar daño, como filtraciones de información privada, discursos de odio, o acoso. Aunque la información base puede ser verdadera, su uso es malicioso.}
}

\newglossaryentry{factchecking}{
	name={Verificación de Hechos (Fact-Checking)},
	description={Proceso sistemático de investigación y verificación de afirmaciones factuales en contenido publicado. Incluye la consulta de fuentes primarias, expertos, y evidencia documental para determinar la veracidad de las afirmaciones.}
}

\newglossaryentry{fraudedigital}{
	name={Fraude Digital},
	description={Conjunto amplio de actividades maliciosas realizadas a través de medios digitales con el objetivo primario de obtener beneficios económicos ilícitos, información personal sensible, o acceso no autorizado a recursos. Incluye estafas en línea, phishing, fraude financiero digital, esquemas Ponzi digitales, estafas de inversión, fraude laboral en línea, y otras formas de engaño que explotan plataformas y tecnologías digitales. A diferencia de las noticias falsas, que buscan influencia social, el fraude digital se enfoca en la obtención directa de beneficios materiales o acceso a activos. Su ``éxito'' se mide por la cantidad de dinero o información obtenida, no por el alcance viral o la influencia en opinión pública.}
}

\newglossaryentry{deepfake}{
	name={Deepfake},
	description={Contenido multimedia (video, audio, imágenes) generado o manipulado usando inteligencia artificial, especialmente técnicas de aprendizaje profundo, para hacer que parezca que alguien dijo o hizo algo que nunca ocurrió realmente.}
}

\newglossaryentry{phishing}{
	name={Phishing},
	description={Técnica de fraude digital que utiliza mensajes electrónicos fraudulentos (correos, SMS, mensajes de redes sociales) que se hacen pasar por comunicaciones legítimas de organizaciones confiables, con el objetivo de obtener información sensible como contraseñas, números de tarjetas de crédito o datos personales.}
}

\newglossaryentry{tokenizacion}{
	name={Tokenización},
	description={Proceso de dividir texto en unidades más pequeñas llamadas tokens, que pueden ser palabras, subpalabras, caracteres o n-gramas. Es el primer paso en el preprocesamiento de texto para análisis computacional.}
}

\newglossaryentry{bolsadepalabras}{
	name={Bolsa de Palabras (Bag of Words)},
	description={Modelo de representación de texto que describe la ocurrencia de palabras en un documento, ignorando el orden y la estructura gramatical. Cada documento se representa como un vector donde cada dimensión corresponde a una palabra del vocabulario y su valor indica la frecuencia de aparición de esa palabra. Es importante para técnicas como TF-IDF y constituye la base para muchos algoritmos de clasificación de texto tradicionales.}
}

\newglossaryentry{stemming}{
	name={Stemming},
	description={Técnica de reducción de palabras a su raíz o stem mediante la eliminación de sufijos. Por ejemplo, ``corriendo'', ``corrió'', ``correr'' se reducen a ``corr''.}
}

\newglossaryentry{lemmatization}{
	name={Lematización},
	description={Proceso más sofisticado que el stemming que reduce las palabras a su forma canónica o lemma, considerando el contexto morfológico y sintáctico. Por ejemplo, ``mejor'' se lematiza a ``bueno''.}
}

\newglossaryentry{stopwords}{
	name={Palabras Vacías (Stop Words)},
	description={Palabras comunes en un idioma que generalmente se filtran durante el preprocesamiento de texto porque no aportan significado semántico significativo. Ejemplos en español: ``el'', ``la'', ``de'', ``que'', ``y''.}
}

\newglossaryentry{embedding}{
	name={Embedding},
	description={Representación vectorial densa de palabras, frases o documentos en un espacio multidimensional donde la distancia entre vectores refleja similitud semántica. Ejemplos incluyen Word2Vec, GloVe, FastText y los embeddings contextuales generados por modelos Transformer.}
}

\newglossaryentry{transformer}{
	name={Transformer},
	description={Arquitectura de red neuronal basada en mecanismos de atención que ha revolucionado el PLN. Introdujo el concepto de atención multi-cabeza y eliminó la necesidad de procesamiento secuencial, permitiendo paralelización eficiente.}
}

\newglossaryentry{attention}{
	name={Atención (Attention)},
	description={Mecanismo que permite a los modelos enfocarse en partes específicas de la entrada al procesar cada elemento. En PLN, permite que el modelo ``atienda'' a palabras relevantes al procesar una palabra específica.}
}

\newglossaryentry{finetuning}{
	name={Fine-tuning},
	description={Proceso de ajustar un modelo pre-entrenado en una tarea específica utilizando un conjunto de datos más pequeño y especializado. Permite aprovechar el conocimiento general del modelo pre-entrenado para tareas particulares.}
}

\newglossaryentry{ngramas}{
	name={n-gramas},
	description={Secuencias contiguas de n elementos (palabras o caracteres) de un texto, que capturan el contexto local y las dependencias secuenciales. En esta tesis, se utilizan n-gramas de palabras como características para la representación TF-IDF.}
}

\newglossaryentry{wordpiece}{
	name={WordPiece},
	description={Algoritmo de tokenización basado en subpalabras utilizado por modelos como BERT y DistilBERT. Divide palabras en unidades subléxicas más pequeñas, permitiendo manejar vocabularios grandes de manera eficiente y procesar palabras fuera del vocabulario mediante la composición de subpalabras conocidas.}
}

\newglossaryentry{preentrenamiento}{
	name={Pre-entrenamiento},
	description={Fase inicial del entrenamiento de un modelo de lenguaje en la que se aprenden representaciones generales del lenguaje utilizando grandes corpus de texto sin etiquetar. En modelos como BERT, incluye tareas como Masked Language Modeling (MLM) y Next Sentence Prediction (NSP).}
}

\newglossaryentry{destilacion}{
	name={Destilación de Conocimiento},
	description={Técnica de compresión de modelos en la que un modelo más pequeño (estudiante) aprende a imitar el comportamiento de un modelo más grande (profesor). DistilBERT utiliza destilación triple combinando pérdida de destilación, pérdida de enmascaramiento y pérdida de similitud coseno.}
}

\newglossaryentry{metaheuristica}{
	name={Metaheurística},
	description={Estrategia de alto nivel para guiar y modificar otras heurísticas con el objetivo de producir soluciones de alta calidad para problemas de optimización. Incluye técnicas como algoritmos genéticos, optimización por enjambre de partículas, y recocido simulado.}
}

\newglossaryentry{hiperparametro}{
	name={Hiperparámetro},
	description={Parámetro de configuración del modelo que se establece antes del entrenamiento y no se aprende durante el proceso de entrenamiento. Ejemplos incluyen la tasa de aprendizaje, el número de épocas, el tamaño del batch, el dropout rate y la regularización L2.}
}

\newglossaryentry{funcionobjetivo}{
	name={Función Objetivo},
	description={Función matemática que define el criterio a optimizar en un problema. En el contexto de esta investigación, representa la métrica de rendimiento del modelo (típicamente F1-Score) que se busca maximizar durante la optimización de hiperparámetros.}
}

\newglossaryentry{espaciobusqueda}{
	name={Espacio de Búsqueda},
	description={Conjunto de todas las posibles configuraciones de hiperparámetros que pueden ser exploradas durante el proceso de optimización. Define los límites y restricciones para cada hiperparámetro.}
}

\newglossaryentry{convergencia}{
	name={Convergencia},
	description={Proceso por el cual un algoritmo de optimización se acerca progresivamente a una solución óptima o cerca del óptimo. Se evalúa observando la estabilización de la función objetivo a lo largo de las iteraciones.}
}

\newglossaryentry{exploracion}{
	name={Exploración},
	description={Capacidad de un algoritmo de búsqueda para investigar regiones no exploradas del espacio de soluciones, evitando quedar atrapado en óptimos locales.}
}

\newglossaryentry{explotacion}{
	name={Explotación},
	description={Capacidad de un algoritmo para refinar y mejorar soluciones prometedoras encontradas, concentrando la búsqueda en regiones del espacio que han mostrado buenos resultados.}
}

\newglossaryentry{operadorcruce}{
	name={Operador de Cruce (Crossover)},
	description={En algoritmos genéticos, operador que combina dos soluciones padres para generar una o más soluciones hijas, heredando características de ambos padres. Implementaciones comunes incluyen cruce de un punto, dos puntos y uniforme.}
}

\newglossaryentry{operadormutacion}{
	name={Operador de Mutación},
	description={En algoritmos genéticos, operador que introduce variabilidad aleatoria en una solución individual modificando uno o más de sus componentes. Previene la convergencia prematura y mantiene diversidad genética en la población.}
}

\newglossaryentry{elitismo}{
	name={Elitismo},
	description={Estrategia en algoritmos evolutivos que garantiza que las mejores soluciones de una generación se preserven y pasen a la siguiente generación sin modificación, asegurando que la calidad de la mejor solución no disminuya.}
}

% Nuevas adiciones: Términos de optimización
\newglossaryentry{kerastuner}{
	name={KerasTuner},
	description={Una biblioteca de software para la optimización de hiperparámetros de modelos de aprendizaje automático, especialmente en TensorFlow, que permite explorar eficientemente diferentes combinaciones de parámetros utilizando estrategias como búsqueda aleatoria, Hyperband y optimización Bayesiana.}
}

\newglossaryentry{learningrate}{
	name={Tasa de Aprendizaje (Learning Rate)},
	description={Un hiperparámetro que determina el tamaño de los pasos que un algoritmo de optimización (como el descenso de gradiente) toma para actualizar los parámetros del modelo en cada iteración. Una tasa de aprendizaje ultra-baja (como 2e-06) se utiliza para una convergencia más controlada y estable.}
}

\newglossaryentry{dropout}{
	name={Dropout},
	description={Una técnica de regularización utilizada en redes neuronales para prevenir el sobreajuste. Consiste en desactivar aleatoriamente una fracción de las neuronas de una capa durante el entrenamiento, obligando al modelo a aprender patrones más robustos y generalizables.}
}

\newglossaryentry{earlystopping}{
	name={Parada Temprana (Early Stopping)},
	description={Un mecanismo de regularización que detiene el proceso de entrenamiento cuando el rendimiento del modelo en un conjunto de validación deja de mejorar durante un número específico de épocas (paciencia). Se utiliza para prevenir el sobreajuste y reducir el tiempo de cómputo.}
}

\newglossaryentry{gapgeneralizacion}{
	name={Gap de Generalización (Generalization Gap)},
	description={La diferencia entre el rendimiento de un modelo en el conjunto de entrenamiento y en un conjunto de validación o prueba. Un gap pequeño (menor a 0.05) indica que el modelo generaliza bien, mientras que un gap grande es un indicador de sobreajuste.}
}

\newglossaryentry{regularizacionl2}{
	name={Regularización L2},
	description={Técnica de regularización que añade una penalización proporcional al cuadrado de la magnitud de los pesos del modelo a la función de pérdida. Ayuda a prevenir el sobreajuste forzando al modelo a utilizar pesos más pequeños y distribuidos, promoviendo soluciones más simples y generalizables.}
}

\newglossaryentry{weightdecay}{
	name={Weight Decay (Decaimiento de Pesos)},
	description={Técnica de regularización que reduce gradualmente la magnitud de los pesos del modelo en cada iteración del entrenamiento. Es equivalente a la regularización L2 en muchos optimizadores y ayuda a prevenir el sobreajuste controlando la capacidad del modelo.}
}

\newglossaryentry{inyeccionruido}{
	name={Inyección de Ruido},
	description={Técnica de regularización que añade ruido aleatorio a las activaciones o pesos del modelo durante el entrenamiento. Actúa como una forma de aumento de datos que mejora la robustez y capacidad de generalización del modelo.}
}

\newglossaryentry{precision}{
	name={Precisión},
	description={Métrica que mide la proporción de predicciones positivas que fueron correctas. Se calcula como: Precisión = TP / (TP + FP). En detección de noticias falsas, indica qué tan confiable es el modelo cuando predice que una noticia es falsa.}
}

\newglossaryentry{recall}{
	name={Sensibilidad (Recall)},
	description={Métrica que mide la proporción de casos positivos reales que fueron correctamente identificados. Se calcula como: Recall = TP / (TP + FN). También conocida como exhaustividad o tasa de verdaderos positivos.}
}

\newglossaryentry{fscore}{
	name={F1-Score},
	description={Media armónica entre precisión y recall. Se calcula como: F1 = 2 × (Precisión × Recall) / (Precisión + Recall). Proporciona una medida equilibrada del rendimiento del clasificador y es la métrica principal utilizada en esta investigación para comparar modelos.}
}

\newglossaryentry{accuracy}{
	name={Exactitud (Accuracy)},
	description={Proporción de predicciones correctas sobre el total de predicciones. Se calcula como: Accuracy = (TP + TN) / (TP + TN + FP + FN). Es una métrica útil cuando las clases están balanceadas.}
}

\newglossaryentry{especificidad}{
	name={Especificidad (Specificity)},
	description={Métrica que mide la capacidad del modelo para identificar correctamente los casos negativos. Se calcula como: Especificidad = TN / (TN + FP). En detección de noticias falsas, mide qué tan bien el modelo identifica noticias reales sin confundirlas con falsas.}
}

\newglossaryentry{aucroc}{
	name={AUC-ROC},
	description={Área bajo la curva ROC (Receiver Operating Characteristic). Mide la capacidad del modelo para distinguir entre clases a través de diferentes umbrales de decisión. Un valor de 1.0 indica un clasificador perfecto, mientras que 0.5 indica rendimiento aleatorio.}
}

\newglossaryentry{matrizconfusion}{
	name={Matriz de Confusión},
	description={Tabla que describe el rendimiento de un modelo de clasificación mostrando las predicciones correctas e incorrectas para cada clase. Permite calcular métricas detalladas de rendimiento como precisión, recall, F1-Score y especificidad.}
}

\newglossaryentry{macroaverage}{
	name={Promedio Macro},
	description={Método de agregación de métricas que calcula la media aritmética simple de las métricas para cada clase, dando el mismo peso a todas las clases independientemente de su frecuencia. Útil para evaluar el rendimiento equilibrado en datasets balanceados.}
}

\newglossaryentry{weightedaverage}{
	name={Promedio Ponderado},
	description={Método de agregación de métricas que calcula la media ponderada de las métricas para cada clase, donde los pesos son proporcionales al número de muestras de cada clase. Refleja el rendimiento global considerando el desbalance de clases.}
}

\newglossaryentry{webscraping}{
	name={Web Scraping},
	description={Técnica de extracción automatizada de datos de sitios web mediante programas que navegan y analizan el contenido HTML de las páginas web para recopilar información específica. En esta investigación se utilizó para extraer 9,000 noticias del portal satírico ``El Deforma''.}
}

\newglossaryentry{dataset}{
	name={Dataset},
	description={Conjunto estructurado de datos utilizado para entrenar, validar y probar modelos de machine learning. En esta investigación, se refiere al corpus de noticias etiquetadas como verdaderas o falsas, que alcanzó más de 61,000 registros únicos.}
}

\newglossaryentry{corpus}{
	name={Corpus},
	description={Colección grande y estructurada de textos utilizados para investigación lingüística o entrenamiento de modelos de PLN. En este contexto, se refiere al conjunto unificado de noticias en español construido integrando cuatro corpus académicos y datos de extracción web.}
}

\newglossaryentry{validacioncruzada}{
	name={Validación Cruzada},
	description={Técnica de evaluación que divide el dataset en múltiples subconjuntos (folds) para entrenar y validar el modelo repetidamente, proporcionando una estimación más robusta del rendimiento. La validación cruzada estratificada preserva la distribución de clases en cada fold.}
}

\newglossaryentry{overfitting}{
	name={Sobreajuste (Overfitting)},
	description={Fenómeno donde un modelo aprende demasiado específicamente los datos de entrenamiento, incluyendo ruido y patrones espurios, perdiendo capacidad de generalización a datos nuevos no vistos durante el entrenamiento. Se manifiesta como un gap grande entre el rendimiento en entrenamiento y validación.}
}

\newglossaryentry{underfitting}{
	name={Subajuste (Underfitting)},
	description={Fenómeno donde un modelo es demasiado simple para capturar la complejidad subyacente de los datos, resultando en bajo rendimiento tanto en entrenamiento como en validación. Indica que el modelo no tiene suficiente capacidad para aprender los patrones relevantes.}
}

\newglossaryentry{bias}{
	name={Sesgo (Bias)},
	description={Error sistemático en las predicciones del modelo que puede deberse a suposiciones erróneas en el algoritmo de aprendizaje, datos de entrenamiento no representativos, o prejuicios inherentes en el dataset. En esta investigación se analizan sesgos temáticos, geográficos y temporales.}
}

\newglossaryentry{varianza}{
	name={Varianza},
	description={Medida de cuánto varían las predicciones del modelo para diferentes conjuntos de entrenamiento. Alta varianza indica que el modelo es sensible a pequeños cambios en los datos de entrenamiento, lo que puede ser una señal de sobreajuste.}
}

\newglossaryentry{divisiondatos}{
	name={División de Datos},
	description={Proceso de particionar el dataset completo en subconjuntos distintos para diferentes propósitos: entrenamiento (70\%), validación (10\%) y prueba (20\%). La división estratificada garantiza que cada subconjunto mantenga la misma proporción de clases que el dataset original.}
}

\newglossaryentry{balanceoclases}{
	name={Balanceo de Clases},
	description={Proceso de ajustar la distribución de ejemplos entre las diferentes clases para evitar sesgos hacia la clase mayoritaria. En esta investigación se logró un balanceo casi perfecto de 49.8\% noticias falsas y 50.2\% noticias reales mediante extracción web dirigida.}
}

\newglossaryentry{contenedorizacion}{
	name={Contenerización},
	description={Técnica de virtualización ligera que empaqueta una aplicación y todas sus dependencias en un contenedor aislado y portable. Docker es la plataforma de contenerización utilizada en esta investigación para garantizar reproducibilidad y facilitar el despliegue.}
}

\newglossaryentry{epoch}{
	name={Época (Epoch)},
	description={Una iteración completa sobre todo el conjunto de datos de entrenamiento. En el entrenamiento del modelo DistilBERT, se utilizaron hasta 30 épocas máximas, aunque el entrenamiento óptimo se detuvo en la época 13 mediante early stopping.}
}

\newglossaryentry{batch}{

	description={Subconjunto de datos procesados simultáneamente en una iteración del algoritmo de entrenamiento. Un tamaño de batch pequeño (como 4, utilizado en esta investigación) proporciona regularización implícita al introducir más ruido en los gradientes.}
}

\newglossaryentry{corpusunificado}{
	name={Corpus Unificado},
	description={Dataset consolidado de más de 61,000 noticias en español construido mediante la integración y estandarización de cuatro corpus académicos reconocidos (Posadas-Durán, Acosta, Tretiakov, Blanco-Fernández) más 9,000 noticias extraídas de ``El Deforma''. Representa una de las contribuciones principales de esta investigación.}
}

\newglossaryentry{españolneutro}{
	name={Español Neutro Multiregional},
	description={Enfoque metodológico que busca desarrollar modelos capaces de procesar múltiples variantes geográficas del español sin estar sesgados hacia una región específica. En esta investigación se logró mediante la integración de corpus de España (70\%), México (20\%) y múltiples países latinoamericanos (10\%).}
}

\newglossaryentry{checkworthy}{
	name={Check-Worthiness},
	description={Propiedad de una afirmación o pieza de contenido que indica si merece ser verificada por un fact-checker. Es una tarea preliminar en sistemas de verificación automática que ayuda a priorizar qué contenido debe ser sometido a verificación exhaustiva.}
}

\newglossaryentry{facedes}{
	name={FakeDeS},
	description={Framework estandarizado para la evaluación de sistemas de detección de noticias falsas en español, establecido por las competencias IberLEF. Define métricas, protocolos de evaluación y datasets de referencia que permiten la comparación directa entre diferentes sistemas.}
}
